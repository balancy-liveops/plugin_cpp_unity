// Balancy WebView Performance Monitor
// Inject this script to monitor WebView performance on real devices

(function() {
    'use strict';
    
    // Performance tracking variables
    let frameCount = 0;
    let lastTime = performance.now();
    let fps = 0;
    let minFps = Infinity;
    let maxFps = 0;
    let avgFps = 0;
    let fpsHistory = [];
    let memoryUsage = 0;
    let frameDrops = 0;
    let lastFrameTime = 0;
    let renderTime = 0;
    let domNodes = 0;
    let networkRequests = 0;
    let failedRequests = 0;
    let totalLoadTime = 0;
    let lastDomCount = performance.now();
    
    // DOM elements
    let performanceOverlay = null;
    let isOverlayVisible = true;
    
    // Configuration
    const PERFORMANCE_CONFIG = {
        maxHistorySize: 60, // Keep last 60 FPS measurements
        updateInterval: 250, // Update display every 250ms (optimized from 100ms)
        domCountInterval: 1000, // Count DOM nodes every 1000ms (optimized)
        frameDropThreshold: 16.67 // Consider frame dropped if > 16.67ms (60fps)
    };
    
    // Create performance overlay
    function createPerformanceOverlay() {
        if (performanceOverlay) return;
        
        performanceOverlay = document.createElement('div');
        performanceOverlay.id = 'balancy-performance-overlay';
        performanceOverlay.style.cssText = `
            position: fixed;
            top: 0;
            left: 0;
            background: rgba(0, 0, 0, 0.8);
            color: #00ff00;
            font-family: 'Courier New', monospace;
            font-size: 12px;
            padding: 8px;
            z-index: 999999;
            border-radius: 0 0 8px 0;
            line-height: 1.2;
            pointer-events: none;
            user-select: none;
            -webkit-user-select: none;
            white-space: pre;
            min-width: 200px;
            transition: opacity 0.3s ease;
        `;
        
        document.body.appendChild(performanceOverlay);
        
        // Toggle visibility on double-tap (top-left corner)
        const toggleArea = document.createElement('div');
        toggleArea.style.cssText = `
            position: fixed;
            top: 0;
            left: 0;
            width: 60px;
            height: 60px;
            z-index: 999998;
            pointer-events: auto;
        `;
        
        let tapCount = 0;
        let tapTimeout = null;
        toggleArea.addEventListener('touchstart', function(e) {
            e.preventDefault();
            tapCount++;
            
            if (tapTimeout) clearTimeout(tapTimeout);
            tapTimeout = setTimeout(() => { 
                tapCount = 0; 
                tapTimeout = null;
            }, 300);
            
            if (tapCount === 2) {
                isOverlayVisible = !isOverlayVisible;
                performanceOverlay.style.opacity = isOverlayVisible ? '1' : '0.1';
                tapCount = 0;
                if (tapTimeout) {
                    clearTimeout(tapTimeout);
                    tapTimeout = null;
                }
            }
        }, { passive: false });
        
        document.body.appendChild(toggleArea);
    }
    
    // Update performance overlay (optimized)
    function updatePerformanceDisplay() {
        if (!performanceOverlay || !isOverlayVisible) return;
        
        const memInfo = performance.memory || {};
        const memUsageMB = memInfo.usedJSHeapSize ? (memInfo.usedJSHeapSize / 1048576).toFixed(1) : 'N/A';
        const status = getPerformanceStatus();
        
        // Pre-calculate values to avoid repeated operations
        const fpsDisplay = fps.toFixed(1);
        const minFpsDisplay = minFps === Infinity ? '0.0' : minFps.toFixed(1);
        const maxFpsDisplay = maxFps.toFixed(1);
        const avgFpsDisplay = avgFps.toFixed(1);
        const renderTimeDisplay = renderTime.toFixed(1);
        
        const displayText = `FPS: ${fpsDisplay} (${minFpsDisplay}-${maxFpsDisplay})
Avg: ${avgFpsDisplay} | Drops: ${frameDrops}
Render: ${renderTimeDisplay}ms
Memory: ${memUsageMB}MB
DOM: ${domNodes} nodes
Net: ${networkRequests}/${failedRequests} (ok/fail)
Load: ${totalLoadTime}ms
${status}`;
        
        performanceOverlay.textContent = displayText;
        
        // Color coding based on performance
        const color = fps < 30 ? '#ff4444' : fps < 50 ? '#ffaa00' : '#00ff00';
        if (performanceOverlay.style.color !== color) {
            performanceOverlay.style.color = color;
        }
    }
    
    // Get performance status
    function getPerformanceStatus() {
        if (fps < 20) return '❌ POOR';
        if (fps < 40) return '⚠️ FAIR';
        if (fps < 55) return '✅ GOOD';
        return '🚀 EXCELLENT';
    }
    
    // Calculate FPS
    function calculateFPS() {
        frameCount++;
        const currentTime = performance.now();
        const deltaTime = currentTime - lastFrameTime;
        
        // Track frame drops
        if (lastFrameTime > 0 && deltaTime > PERFORMANCE_CONFIG.frameDropThreshold) {
            frameDrops++;
        }
        
        lastFrameTime = currentTime;
        
        // Calculate FPS every second
        if (currentTime - lastTime >= 1000) {
            fps = (frameCount * 1000) / (currentTime - lastTime);
            
            // Update min/max FPS
            if (fps < minFps) minFps = fps;
            if (fps > maxFps) maxFps = fps;
            
            // Update FPS history
            fpsHistory.push(fps);
            if (fpsHistory.length > PERFORMANCE_CONFIG.maxHistorySize) {
                fpsHistory.shift();
            }
            
            // Calculate average FPS
            avgFps = fpsHistory.reduce((sum, f) => sum + f, 0) / fpsHistory.length;
            
            // Reset counters
            frameCount = 0;
            lastTime = currentTime;
        }
        
        requestAnimationFrame(calculateFPS);
    }
    
    // Monitor render time (optimized)
    function measureRenderTime() {
        const startTime = performance.now();
        
        // Use requestAnimationFrame to measure actual render completion
        requestAnimationFrame(() => {
            renderTime = performance.now() - startTime;
        });
    }
    
    // Count DOM nodes (optimized - called less frequently)
    function countDOMNodes() {
        const currentTime = performance.now();
        if (currentTime - lastDomCount >= PERFORMANCE_CONFIG.domCountInterval) {
            domNodes = document.getElementsByTagName('*').length;
            lastDomCount = currentTime;
        }
    }
    
    // Monitor network requests (optimized)
    function monitorNetworkRequests() {
        // Monitor fetch requests
        if (window.fetch && !window.fetch._balancyMonitored) {
            const originalFetch = window.fetch;
            window.fetch = function(...args) {
                networkRequests++;
                return originalFetch.apply(this, args)
                    .catch(error => {
                        failedRequests++;
                        throw error;
                    });
            };
            window.fetch._balancyMonitored = true;
        }
        
        // Monitor XMLHttpRequest
        if (window.XMLHttpRequest && !XMLHttpRequest.prototype._balancyMonitored) {
            const originalXHROpen = XMLHttpRequest.prototype.open;
            XMLHttpRequest.prototype.open = function() {
                networkRequests++;
                const xhr = this;
                
                // Add error handler only once
                if (!xhr._balancyErrorHandlerAdded) {
                    const originalOnError = xhr.onerror;
                    xhr.onerror = function() {
                        failedRequests++;
                        if (originalOnError) originalOnError.apply(this, arguments);
                    };
                    xhr._balancyErrorHandlerAdded = true;
                }
                
                return originalXHROpen.apply(this, arguments);
            };
            XMLHttpRequest.prototype._balancyMonitored = true;
        }
    }
    
    // Calculate page load time
    function calculateLoadTime() {
        if (document.readyState === 'complete') {
            const timing = performance.timing;
            if (timing && timing.loadEventEnd && timing.navigationStart) {
                totalLoadTime = timing.loadEventEnd - timing.navigationStart;
            }
        } else {
            const handleLoad = () => {
                setTimeout(calculateLoadTime, 100);
                window.removeEventListener('load', handleLoad);
            };
            window.addEventListener('load', handleLoad);
        }
    }
    
    // Monitor memory leaks (optimized)
    function monitorMemoryLeaks() {
        if (!performance.memory) return;
        
        const memoryCheckInterval = setInterval(() => {
            const memInfo = performance.memory;
            const currentUsage = memInfo.usedJSHeapSize;
            
            // Simple memory leak detection - if memory grows continuously
            if (memoryUsage > 0 && currentUsage > memoryUsage * 1.5 && fpsHistory.length > 30) {
                console.warn('Balancy: Potential memory leak detected:', {
                    current: (currentUsage / 1048576).toFixed(1) + 'MB',
                    previous: (memoryUsage / 1048576).toFixed(1) + 'MB'
                });
            }
            
            memoryUsage = currentUsage;
        }, 5000);
        
        // Clear interval on page unload
        window.addEventListener('beforeunload', () => {
            clearInterval(memoryCheckInterval);
        });
    }
    
    // Monitor scroll performance (optimized)
    function monitorScrollPerformance() {
        let scrollStartTime = 0;
        let isScrolling = false;
        let scrollEndTimer = null;
        
        const handleScroll = () => {
            if (!isScrolling) {
                scrollStartTime = performance.now();
                isScrolling = true;
            }
            
            if (scrollEndTimer) clearTimeout(scrollEndTimer);
            scrollEndTimer = setTimeout(() => {
                const scrollDuration = performance.now() - scrollStartTime;
                if (scrollDuration > 100) {
                    console.warn('Balancy: Slow scroll detected:', scrollDuration + 'ms');
                }
                isScrolling = false;
                scrollEndTimer = null;
            }, 150);
        };
        
        window.addEventListener('scroll', handleScroll, { passive: true });
        
        // Cleanup on page unload
        window.addEventListener('beforeunload', () => {
            window.removeEventListener('scroll', handleScroll);
            if (scrollEndTimer) clearTimeout(scrollEndTimer);
        });
    }
    
    // Log performance data for debugging
    function logPerformanceData() {
        const perfData = {
            fps: parseFloat(fps.toFixed(1)),
            minFps: minFps === Infinity ? 0 : parseFloat(minFps.toFixed(1)),
            maxFps: parseFloat(maxFps.toFixed(1)),
            avgFps: parseFloat(avgFps.toFixed(1)),
            frameDrops: frameDrops,
            renderTime: parseFloat(renderTime.toFixed(1)),
            memoryUsage: performance.memory ? parseFloat((performance.memory.usedJSHeapSize / 1048576).toFixed(1)) : null,
            domNodes: domNodes,
            networkRequests: networkRequests,
            failedRequests: failedRequests,
            loadTime: totalLoadTime,
            timestamp: Date.now(),
            userAgent: navigator.userAgent
        };
        
        // Send to native app if available
        //if (window.balancy && typeof window.balancy.postMessage === 'function') {
        //    try {
        //        window.balancy.postMessage('//performance_data:' + JSON.stringify(perfData));
        //    } catch (e) {
        //        console.warn('Balancy: Failed to send performance data to native:', e);
        //    }
        //}
        
        console.log('Balancy WebView Performance Data:', perfData);
        return perfData;
    }
    
    // Initialize performance monitoring
    function initPerformanceMonitoring() {
        console.log('Balancy WebView Performance Monitor initialized');
        
        createPerformanceOverlay();
        calculateLoadTime();
        monitorNetworkRequests();
        monitorMemoryLeaks();
        monitorScrollPerformance();
        
        // Start FPS calculation
        requestAnimationFrame(calculateFPS);
        
        // Update display periodically (optimized intervals)
        const displayUpdateInterval = setInterval(() => {
            if (isOverlayVisible) {
                measureRenderTime();
                countDOMNodes();
                updatePerformanceDisplay();
            }
        }, PERFORMANCE_CONFIG.updateInterval);
        
        // Log performance data every 10 seconds
        const logInterval = setInterval(logPerformanceData, 10000);
        
        // Cleanup on page unload
        window.addEventListener('beforeunload', () => {
            clearInterval(displayUpdateInterval);
            clearInterval(logInterval);
        });
        
        // Expose performance data to global scope
        window.balancyPerformance = {
            getPerformanceData: logPerformanceData,
            toggleOverlay: () => {
                isOverlayVisible = !isOverlayVisible;
                if (performanceOverlay) {
                    performanceOverlay.style.opacity = isOverlayVisible ? '1' : '0.1';
                }
            },
            resetStats: () => {
                frameDrops = 0;
                minFps = Infinity;
                maxFps = 0;
                fpsHistory = [];
                networkRequests = 0;
                failedRequests = 0;
                console.log('Balancy: Performance stats reset');
            },
            isVisible: () => isOverlayVisible,
            getConfig: () => ({ ...PERFORMANCE_CONFIG })
        };
    }
    
    // Initialize when DOM is ready
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initPerformanceMonitoring);
    } else {
        // Use setTimeout to ensure all other scripts have loaded
        setTimeout(initPerformanceMonitoring, 0);
    }
    
})();
