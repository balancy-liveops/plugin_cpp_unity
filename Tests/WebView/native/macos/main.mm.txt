extern "C" void UnitySendMessage(const char*, const char*, const char*) {}
// Include production implementation so private native controller lifetimes are exercised.
#import "../../../../WebView/Plugins/macOS/BalancyWebviewMac.mm"
static int failures, checks, messages;
static bool loadFinished, loadSucceeded;
static void Check(const char *name, bool ok) { checks++; printf("%s %s\n",ok ? "PASS" : "FAIL",name); if(!ok) failures++; }
static void Message(const char *) { messages++; }
static void LoadCompleted(bool success) { loadFinished = true; loadSucceeded = success; }
static bool WaitForLoad(void) {
 NSDate *deadline = [NSDate dateWithTimeIntervalSinceNow:5.0];
 while (!loadFinished && deadline.timeIntervalSinceNow > 0) {
  [[NSRunLoop currentRunLoop] runUntilDate:[NSDate dateWithTimeIntervalSinceNow:.01]];
 }
 return loadFinished && loadSucceeded;
}
@interface LateMessage : WKScriptMessage
@property(nonatomic,strong) WKWebView *source;
@end
@implementation LateMessage
- (NSString *)name { return @"BalancyWebView"; }
- (id)body { return @"late"; }
- (WKWebView *)webView { return self.source; }
@end
int main() {
 @autoreleasepool {
  [NSApplication sharedApplication]; [NSApp setActivationPolicy:NSApplicationActivationPolicyAccessory];
  _messageCallback = Message;
  NSString *encodedDirectory = [NSTemporaryDirectory() stringByAppendingPathComponent:@"Balancy WebView URL Test %20 # ü"];
  [[NSFileManager defaultManager] createDirectoryAtPath:encodedDirectory withIntermediateDirectories:YES attributes:nil error:nil];
  NSString *encodedShell = [encodedDirectory stringByAppendingPathComponent:@"shell page %20 # ü.html"];
  [@"<!doctype html><title>Balancy</title>" writeToFile:encodedShell atomically:YES encoding:NSUTF8StringEncoding error:nil];
  NSString *encodedShellURL = [NSURL fileURLWithPath:encodedShell].absoluteString;
  Check("fixture URL escapes spaces, percent and hash", [encodedShellURL containsString:@"%20"] && [encodedShellURL containsString:@"%2520"] && [encodedShellURL containsString:@"%23"]);
  _loadCompletedCallback = LoadCompleted;
  loadFinished = false; loadSucceeded = false;
  Check("persistent encoded file URL starts", _balancyPrepareWebView([encodedShellURL UTF8String]));
  Check("persistent encoded file URL loads", WaitForLoad());
  _loadCompletedCallback = NULL;
  _balancyCloseWebView();

  _loadCompletedCallback = LoadCompleted;
  loadFinished = false; loadSucceeded = false;
  Check("classic encoded file URL starts", _balancyOpenWebViewWithSize([encodedShellURL UTF8String], 320, 200));
  Check("classic encoded file URL loads", WaitForLoad());
  _loadCompletedCallback = NULL;
  _balancyCloseWebView();

  BalancyEmbeddedWebViewController *encodedEmbedded = [[BalancyEmbeddedWebViewController alloc] initWithWidth:64 height:64];
  _loadCompletedCallback = LoadCompleted;
  loadFinished = false; loadSucceeded = false;
  Check("embedded encoded file URL starts", [encodedEmbedded loadURL:encodedShellURL]);
  Check("embedded encoded file URL loads", WaitForLoad());
  _loadCompletedCallback = NULL;
  [encodedEmbedded close];
  encodedEmbedded = nil;
  [[NSFileManager defaultManager] removeItemAtPath:encodedDirectory error:nil];
  // Exercise exported entry points: ordinary and prepared windows must agree.
  for (NSValue *value in @[[NSValue valueWithSize:NSMakeSize(360,640)],
                           [NSValue valueWithSize:NSMakeSize(640,360)]]) {
   NSSize size = value.sizeValue;
   Check("classic sized open starts", _balancyOpenWebViewWithSize("about:blank", size.width, size.height));
   NSSize classic = _sharedController.webView.bounds.size;
   Check("classic matches requested viewport", NSEqualSizes(classic, size));
   _balancyCloseWebView();
   Check("sized persistent prepare starts", _balancyPrepareWebViewWithSize("about:blank", size.width, size.height));
   Check("persistent matches classic viewport", NSEqualSizes(_sharedController.webView.bounds.size, classic));
   Check("sized preparation stays hidden", !_sharedController.window.visible);
   NSSize currentGameView = NSMakeSize(size.height, size.width);
   _balancySetWindowSize(currentGameView.width, currentGameView.height);
   Check("prepared shell adopts current Game View viewport", NSEqualSizes(_sharedController.webView.bounds.size, currentGameView));
   _balancyShowWebView();
   Check("show preserves current Game View viewport", NSEqualSizes(_sharedController.webView.bounds.size, currentGameView));
   _balancyCloseWebView();
  }
  Check("legacy prepare entry point remains callable", _balancyPrepareWebView("about:blank"));
  Check("legacy prepare retains default size", NSEqualSizes(_sharedController.webView.bounds.size, NSMakeSize(800,600)));
  _balancyCloseWebView();
  _balancyPrepareWebView("about:blank");
  BalancyWebViewController *emergency = _sharedController;
  _balancyShowWebView();
  Check("emergency initially absent", emergency.emergencyExitButton == nil);
  [emergency showEmergencyExitButton];
  Check("triple click reveals emergency", emergency.emergencyExitButton != nil && !emergency.emergencyExitButton.hidden);
  [emergency emergencyExitButtonClicked:nil];
  Check("emergency exit sends close", messages == 1);
  messages = 0;
  Check("exit removes emergency button and timer", emergency.emergencyExitButton == nil && emergency.emergencyExitHideTimer == nil);
  [emergency hideForPersistentMode]; [emergency startShowAnimation];
  Check("reopen does not inherit emergency", emergency.emergencyExitButton == nil);
  [emergency showEmergencyExitButton]; [emergency hideForPersistentMode]; [emergency startShowAnimation];
  Check("regular hide resets emergency and timer", emergency.emergencyExitButton == nil && emergency.emergencyExitHideTimer == nil);
  _balancySetEmergencyExitEnabled(false); [emergency showEmergencyExitButton];
  Check("disabled emergency cannot appear", emergency.emergencyExitButton == nil);
  _balancySetEmergencyExitEnabled(true); [emergency showEmergencyExitButton];
  Check("reenabled emergency can appear", emergency.emergencyExitButton != nil);
  _balancyCloseWebView(); emergency = nil;
  NSHashTable *retired = [NSHashTable weakObjectsHashTable];
  for(int i=0;i<20;i++) {
   @autoreleasepool {
    BalancyWebViewController *c = [[BalancyWebViewController alloc] initWithSize:NSMakeSize(100,100)];
    [retired addObject:c];
    Check("window ownership stays with ARC", !c.window.releasedWhenClosed);
    [c preparePersistentShellLoad];
    Check("prepare hidden", !c.window.visible);
    [c setShowDelay:0];[c setAnimationDuration:0];[c startShowAnimation];
    Check("zero duration shows synchronously", c.window.alphaValue == 1);
    c.showDelay=.1;[c startShowAnimation];[c hideForPersistentMode];
    Check("hide disables visible window", !c.window.visible);
    WKWebView *view = c.webView; WKUserContentController *content = c.userContentController;
    [c close];[c close];
    Check("close clears native references", c.webView == nil && c.userContentController == nil);
    Check("close detaches delegates and hierarchy", view.navigationDelegate == nil && view.UIDelegate == nil && view.superview == nil);
    Check("send after close rejected", ![c sendMessage:@"closed"]);
    LateMessage *late = [LateMessage new];late.source=view;
    [c userContentController:content didReceiveScriptMessage:late];
    Check("late script message ignored", messages == 0);
   }
   [[NSRunLoop currentRunLoop] runUntilDate:[NSDate dateWithTimeIntervalSinceNow:.05]];
  }
  for(int i=0;i<5;i++) {
   @autoreleasepool {
    BalancyEmbeddedWebViewController *c = [[BalancyEmbeddedWebViewController alloc] initWithWidth:64 height:64];
    [retired addObject:c];
    Check("embedded window ownership stays with ARC", !c.offscreenWindow.releasedWhenClosed);
    [c close];[c close];
    Check("embedded close releases pixel buffer and graphics context", c.pixelBuffer == nil && c.persistentContext == nil);
    Check("embedded close releases timer and window", c.renderTimer == nil && c.offscreenWindow == nil);
    Check("embedded close releases view and handlers", c.webView == nil && c.userContentController == nil);
   }
  }
  [[NSRunLoop currentRunLoop] runUntilDate:[NSDate dateWithTimeIntervalSinceNow:1]];
  Check("retired window controllers released", retired.allObjects.count == 0);
  printf("RESULT failures=%d checks=%d\n",failures,checks);
  return failures ? 1 : 0;
 }
}
