const balancy = {};

(function() {
    const pendingRequests = {};
    let requestCounter = 0;

    const RequestAction = {
        None: 0,
        GetProfile: 1,
        SetProfile: 2,
        GetDocument: 3,
        GetLocalization: 10,
        GetImageUrl: 11,
        GetInfo: 12,
        CanBuyGroupOffer: 13,

        GetBattlePassProgress: 20,

        BuyOffer: 101,
        BuyGroupOffer: 102,
        BuyShopSlot: 103,
        BattlePassClaim: 104,

        CloseWindow: 200,
        BalancyIsReady: 201,
        CustomMessage: 1000,
    };

    const InfoType = {
        None: 0,
        OfferPrice: 1,
        OfferGroupPrice: 2,
        TimeLeft: 3,
        CustomPrice: 9,
        Custom: 10
    };

    const NotificationType = {
        OnNewEventActivated: 100,
        OnEventDeactivated: 101,
        OnNewOfferActivated: 102,
        OnOfferDeactivated: 103,
        OnNewOfferGroupActivated: 104,
        OnOfferGroupDeactivated: 105,

        OnShopSlotWasPurchased: 120,
        OnOfferWasPurchased: 121,
        OnOfferGroupWasPurchased: 122,
    };

    balancy.findDomElement = (dataId, parentElement) => {
        const element = (parentElement ?? document).querySelector(`[data-id=${dataId}]`);
        return element instanceof HTMLElement ? element : null;
    }

    balancy.getSystemProfileValue = (path) => {
        return balancy.sendRequest(null, RequestAction.GetProfile, {profile: "UnnyProfile", path});
    };

    balancy.getProfileValue = (profile, path) => {
        return balancy.sendRequest(null, RequestAction.GetProfile, {profile, path});
    };

    balancy.closeView = (source) => {
        return balancy.sendRequest(null, RequestAction.CloseWindow, {id: source});
    };

    balancy.getDocumentValue = (id, depth) => {
        return balancy.sendRequest(null, RequestAction.GetDocument, {id, depth}).then(result => {
            return JSON.parse(result);
        });
    };

    balancy.getBattlePassProgress = () => {
        return balancy.sendRequest(null, RequestAction.GetBattlePassProgress, {id: window.balancyViewOwner.unnyIdGameEvent}).then(result => {
            return JSON.parse(result);
        });
    };

    balancy.claimBattlePassReward = (lineId, index) => {
        return balancy.sendRequest(null, RequestAction.BattlePassClaim, {id: window.balancyViewOwner.unnyIdGameEvent, lineId, index}).then(response => {
            return response && response === 'ok';
        });
    };

    balancy.getLocalizedText = async function(key) {
        return balancy.sendRequest(null, RequestAction.GetLocalization, {key});
    };

    balancy.getImageUrl = async function(id) {
        return balancy.sendRequest(null, RequestAction.GetImageUrl, {id});
    };

    balancy.sendIsReady = () => {
        return balancy.sendRequest(null, RequestAction.BalancyIsReady, {id: 'none'});
    };

    balancy.delayIsReady = () => {
        balancy.delayedIsReady = true;
    };

    balancy.canBuyGroupOffer = (index) => {
        return balancy.sendRequest(null, RequestAction.CanBuyGroupOffer, {instanceId: window.balancyViewOwner.instanceId, index});
    };

    async function prepareSprite(element, id) {
        const imageUri = await balancy.getImageUrl(id);
        if (imageUri) {
            if (element.tagName === 'IMG')
                element.src = imageUri;
            else
                element.style.borderImageSource = `url('${imageUri}')`;
            element.classList.add('balancy-image-prepared');
        }
    }

    balancy.setImage = function(element, sprite) {
        element.setAttribute('data-image-id', sprite.id);
        element.classList.remove('balancy-image-prepared');
        prepareSprite(element, sprite.id);
    }

    balancy.postMessage = function(message) {
        // iOS/macOS WebKit implementation
        if (window.webkit && window.webkit.messageHandlers && window.webkit.messageHandlers.BalancyWebView) {
            if (typeof message !== 'string') message = JSON.stringify(message);
            window.webkit.messageHandlers.BalancyWebView.postMessage(message);
            return true;
        }

        // Android JavascriptInterface implementation
        if (window.BalancyWebView && window.BalancyWebView.sendMessageToUnity) {
            if (typeof message !== 'string') message = JSON.stringify(message);
            window.BalancyWebView.sendMessageToUnity(message);
            return true;
        }

        // Fallback to postMessage for web views that support it
        if (window.parent && window.parent !== window && window.parent.postMessage) {
            window.parent.postMessage({
                source: 'balancy-view',
                type: 'message',
                payload: message,
                timestamp: Date.now()
            }, '*');
            return true;
        }

        console.warn('BalancyWebView: No message handler available for platform');
        if (window.BalancyWebView && window.BalancyWebView.log) {
            window.BalancyWebView.log('warn', 'No message handler available for platform');
        }

        return false;
    };

    balancy._receiveMessageFromUnity = function(message) {
        if (window.BalancyWebView && window.BalancyWebView.log) {
            window.BalancyWebView.log('info', 'Received message from Unity: ' + message.substring(0, 100) + '...');
        } else {
            console.log('BalancyWebView: Received message from Unity:', message.substring(0, 100) + '...');
        }

        balancy.handleResponse(message);
    };

    balancy.sendRequest = function(sender, action, params = {}) {
        return new Promise((resolve, reject) => {
            const requestId = (requestCounter++).toString();

            pendingRequests[requestId] = {
                resolve: resolve,
                reject: reject,
                timestamp: Date.now()
            };

            const message = {
                type: 'request',
                sender,
                id: requestId,
                action: action,
                params: params
            };

            balancy.postMessage(JSON.stringify(message));
        });
    };

    async function localizeAllElements() {
        const elements = document.querySelectorAll('[data-text-type="localized"]');

        const localizationPromises = Array.from(elements).map(async (element) => {
            if (element.classList.contains('balancy-text-prepared'))
                return;

            const key = element.getAttribute('data-localization-key');
            if (key) {
                const localizedText = await balancy.getLocalizedText(key);
                element.textContent = localizedText;
            }

            element.classList.add('balancy-text-prepared');
        });

        await Promise.all(localizationPromises);

        document.dispatchEvent(new CustomEvent('balancy-localization-complete'));
    }

    async function prepareAllFonts() {
        const elements = document.querySelectorAll('[data-font-id]');
        const fontsPromises = Array.from(elements).map(async (element) => {
            if (element.classList.contains('balancy-font-prepared'))
                return;

            const name = element.getAttribute('data-font-name');
            const id = element.getAttribute('data-font-id');
            const fontUri = await balancy.getImageUrl(id);
            balancy.injectFontFace(name, fontUri);
        });

        await Promise.all(fontsPromises);

        document.dispatchEvent(new CustomEvent('balancy-fonts-complete'));
    }

    async function prepareAllSprites() {
        const elements = document.querySelectorAll('[data-image-id]');
        const imagesPromises = Array.from(elements).map(async (element) => {
            if (element.classList.contains('balancy-image-prepared'))
                return;

            const id = element.getAttribute('data-image-id');
            await prepareSprite(element, id);
        });

        await Promise.all(imagesPromises);

        document.dispatchEvent(new CustomEvent('balancy-images-complete'));
    }

    function refreshButtonAvailability(button) {
        const actionIdStr = button.getAttribute('data-button-action');
        if (!actionIdStr)
            return;
        const actionId = parseInt(actionIdStr, 10);

        switch (actionId) {
            case RequestAction.BuyGroupOffer:
                const index = parseInt(button.getAttribute('data-index')) || 0;
                balancy.canBuyGroupOffer(index).then(result => {
                    console.log(">>>>balancy.canBuyGroupOffer ", result);
                    button.disabled = !result;
                });
                break;
        }
    }

    function balancyButtonClicked(button) {
        const actionIdStr = button.getAttribute('data-button-action');
        if (!actionIdStr) {
            console.warn('Button missing data-button-action attribute', button);
            return;
        }

        const actionId = parseInt(actionIdStr, 10);
        if (isNaN(actionId) || actionId === RequestAction.None) {
            console.warn('Invalid action ID:', actionIdStr, button);
            return;
        }

        const senderId = button.id || 'unknown-button';
        const paramsAttr = button.getAttribute('data-button-params');
        let paramsJson = {};

        if (paramsAttr && paramsAttr.trim() !== '') {
            try {
                paramsJson = JSON.parse(paramsAttr);
            } catch (error) {
                console.error('Error parsing button params JSON:', error, 'Raw params:', paramsAttr);
                // Dispatch error event but don't return - continue with empty params
                button.dispatchEvent(new CustomEvent('balancyButtonError', {
                    detail: { actionId, error: 'Invalid JSON in data-button-params', paramsAttr }
                }));
                paramsJson = {};
            }
        }

        switch (actionId) {
            case RequestAction.BuyOffer:
                paramsJson.instanceId = window.balancyViewOwner.instanceId;
                break;
            case RequestAction.BuyGroupOffer:
                paramsJson.instanceId = window.balancyViewOwner.instanceId;
                paramsJson.index = parseInt(button.getAttribute('data-index')) || 0;
                break;
            case RequestAction.BuyShopSlot:
                paramsJson.slotId = button.getAttribute('data-index') || 0;
                break;
        }

        const originalDisabled = button.disabled;
        button.disabled = true;
        button.classList.add('balancy-button-processing');

        balancy.sendRequest(senderId, actionId, paramsJson)
            .then(result => {
                console.log(`Button action ${actionId} executed successfully`, result);
                button.dispatchEvent(new CustomEvent('balancyButtonResponse', {
                    detail: { actionId, result, success: true, senderId }
                }));
            })
            .catch(error => {
                console.error(`Error executing button action ${actionId}:`, error);
                button.dispatchEvent(new CustomEvent('balancyButtonResponse', {
                    detail: { actionId, error: error.message, success: false, senderId }
                }));
            })
            .finally(() => {
                button.disabled = originalDisabled;
                button.classList.remove('balancy-button-processing');
            });
    }

    function prepareAllButtons() {
        const elements = document.querySelectorAll('[data-button-action]');
        let preparedCount = 0;

        elements.forEach(element => {
            if (element.classList.contains('balancy-button-prepared'))
                return;

            if (element._balancyClickHandler)
                element.removeEventListener('click', element._balancyClickHandler);

            element._balancyClickHandler = function(event) {
                event.preventDefault();
                event.stopPropagation();
                balancyButtonClicked(element);
            };

            refreshButtonAvailability(element);

            element.addEventListener('click', element._balancyClickHandler);

            element.classList.add('balancy-button-prepared');
            preparedCount++;
        });

        document.dispatchEvent(new CustomEvent('balancy-buttons-complete', {
            detail: { preparedCount, totalFound: elements.length }
        }));
    }

    function prepareAllTexts() {
        const elements = document.querySelectorAll('[data-text-type="dynamic"]');

        const promises = [];

        elements.forEach(element => {
            if (element.classList.contains('balancy-text-prepared'))
                return;

            element.classList.add('balancy-text-prepared');

            const senderId = element.id || 'unknown-text';
            let type = parseInt(element.getAttribute('data-info-type')) || 0;
            const index = parseInt(element.getAttribute('data-index')) || 0;
            const custom = element.getAttribute('data-custom') || '';
            let productId = element.getAttribute('data-product-id') || '';
            let instanceId = '';

            if (type === InfoType.OfferPrice) {
                productId = window.balancyViewOwner.productId;
                type = InfoType.CustomPrice;
            } else if (type === InfoType.OfferGroupPrice) {
                instanceId = window.balancyViewOwner.instanceId;
            }

            if (type === InfoType.TimeLeft) {
                const timerElement = element;
                function setTimeValue() {
                    timerElement.textContent = balancy.formatTime(balancy.getTimeLeft());
                }
                setInterval(setTimeValue, 1000);
                setTimeValue();
                return;
            }

            const promise = balancy.sendRequest(senderId, RequestAction.GetInfo, {
                type,
                productId,
                index,
                instanceId,
                custom,
            }).then(result => {
                try {
                    const formatStr = element.getAttribute("data-text-format");
                    element.textContent = balancy.formatDataTemplate(formatStr, result);
                } catch (error) {
                    console.error('Error formatting text:', error, 'Result:', result);
                }
            });

            promises.push(promise);
        });


        Promise.all(promises).then(() => {
            document.dispatchEvent(new CustomEvent('balancy-text-complete'));
        });
    }

    balancy.notificationReceived = function(data) {
        if (!data || typeof data !== 'object')
            return;

        console.log(">>>notificationReceived ", data);
        switch (data.type) {
            case NotificationType.OnOfferDeactivated:
            case NotificationType.OnOfferGroupDeactivated:
                if (window.balancyViewOwner.instanceId === data.id)
                    balancy.closeView('Offer deactivated');
                break;
            case NotificationType.OnOfferGroupWasPurchased:
                const elements = document.querySelectorAll(`[data-button-action="${RequestAction.BuyGroupOffer}"]`);
                elements.forEach(refreshButtonAvailability);
                break;
        }
    }

    balancy.initResponseHandler = function() {
        if (balancy._responseHandlerInitialized) return;

        balancy._responseHandlerInitialized = true;

        balancy.handleResponse = function(responseJson) {
            try {
                const response = JSON.parse(responseJson);

                if (response.type === 'response') {
                    const request = pendingRequests[response.id];
                    if (request) {
                        delete pendingRequests[response.id];
                        if (response.error)
                            request.reject(new Error(response.error));
                        else
                            request.resolve(response.result);
                    }
                } else {
                    if (response.type === 'notification') {
                        const data = response.data ? JSON.parse(response.data) : null;
                        balancy.notificationReceived(data);
                    }
                }
            } catch (error) {
                console.error('Error handling response:', error);
            }
        };

        window.addEventListener('message', function(event) {
            if (event.source !== window.parent)
                return;

            if (event.source === window)
                return;

            const data = event.data;
            if (data && typeof data === 'object') {
                if (data.payload)
                    balancy.handleResponse(data.payload);
            }
        });

        (function waitForAllBalancyEvents() {
            const requiredEvents = new Set([
                'balancy-buttons-complete',
                'balancy-localization-complete',
                'balancy-text-complete',
                'balancy-images-complete',
                'balancy-fonts-complete',
            ]);

            const receivedEvents = new Set();

            function checkAllReceived(event) {
                receivedEvents.add(event.type);
                console.log(':: STATUS :: ' + event.type);
                if ([...requiredEvents].every(e => receivedEvents.has(e))) {
                    console.log("ALL IS READY 1");
                    window.dispatchEvent(new CustomEvent("balancy-ready"));

                    if (!balancy.delayedIsReady)
                        balancy.sendIsReady();
                }
            }

            requiredEvents.forEach(eventName => {
                document.addEventListener(eventName, checkAllReceived, { once: true });
            });
        })();

        localizeAllElements();
        prepareAllFonts();
        prepareAllSprites();
        prepareAllButtons();
        prepareAllTexts();
    };

    balancy.getTimeLeft = function() {
        const timeNow = Math.floor(Date.now() / 1000);
        const timeSpend = timeNow - window.balancySettings.launchTime;
        return window.balancySettings.secondsLeft - timeSpend;
    }

    balancy.formatTime = function (seconds) {
        if (seconds > 3600 * 24 * 365) // 1 year
            return 'oo';

        if (window.customFormatTime)
            return window.customFormatTime(seconds);

        const days = Math.floor(seconds / 86400);
        const hours = Math.floor((seconds % 86400) / 3600);
        const minutes = Math.floor((seconds % 3600) / 60);
        const secs = seconds % 60;

        if (days > 0) {
            if (hours > 0)
                return `${days}d ${hours}h`;
            return `${days}d`;
        } else
            return `${hours.toString().padStart(2, '0')}:${minutes.toString().padStart(2, '0')}:${secs.toString().padStart(2, '0')}`;
    }

    balancy.formatDataTemplate = function(formatStr, data) {
        function getNestedValue(obj, path) {
            return path.split('.').reduce((current, key) => {
                return current && current[key] !== undefined ? current[key] : undefined;
            }, obj);
        }

        if (formatStr.includes('{') && formatStr.includes('}')) {
            return formatStr.replace(/\{([^}]+)\}/g, (match, path) => {
                const value = getNestedValue(data, path.trim());
                return value !== undefined ? String(value) : match;
            });
        }
    }

    balancy.injectFontFace = function(fontName, localFontPath) {
        const style = document.createElement('style');
        style.type = 'text/css';

        style.textContent = `
            @font-face {
                font-family: '${fontName}';
                src: url('${localFontPath}') format('truetype');
            }
        `;
        document.head.appendChild(style);
    }

    balancy.initResponseHandler();
})();

window.balancy = balancy;
