    // Универсальные JavaScript оптимизации
    function applyGameUIBehavior() {
        // Отключение контекстного меню
        document.addEventListener('contextmenu', function(e) {
            e.preventDefault();
            e.stopPropagation();
            return false;
        }, { passive: false, capture: true });
        
        // Отключение выделения текста
        document.addEventListener('selectstart', function(e) {
            e.preventDefault();
            e.stopPropagation();
            return false;
        }, { passive: false, capture: true });
        
        // Отключение drag and drop
        document.addEventListener('dragstart', function(e) {
            e.preventDefault();
            e.stopPropagation();
            return false;
        }, { passive: false, capture: true });
        
        // iOS специфичные оптимизации
        if (/iPad|iPhone|iPod/.test(navigator.userAgent)) {
            console.log('[BalancyWebView] Applying iOS-specific optimizations');
            
            // Отключение bounce эффекта
            document.addEventListener('touchmove', function(e) {
                if (!['INPUT', 'TEXTAREA'].includes(e.target.tagName)) {
                    e.preventDefault();
                }
            }, { passive: false });
            
            // Предотвращение zoom при double tap
            let lastTouchEnd = 0;
            document.addEventListener('touchend', function(event) {
                const now = Date.now();
                if (now - lastTouchEnd <= 300) {
                    event.preventDefault();
                }
                lastTouchEnd = now;
            }, { passive: false });
            
            // Отключение callout
            document.addEventListener('touchstart', function(e) {
                e.target.style.webkitTouchCallout = 'none';
            }, { passive: true });
        }
        
        // Android специфичные оптимизации
        if (/Android/.test(navigator.userAgent)) {
            console.log('[BalancyWebView] Applying Android-specific optimizations');
            
            // Предотвращение системных жестов по краям экрана
            document.addEventListener('touchstart', function(e) {
                if (e.touches[0].clientX < 20 || 
                    e.touches[0].clientX > window.innerWidth - 20 ||
                    e.touches[0].clientY < 20 ||
                    e.touches[0].clientY > window.innerHeight - 20) {
                    e.preventDefault();
                }
            }, { passive: false });
            
            // Отключение pull-to-refresh
            let startY = 0;
            document.addEventListener('touchstart', function(e) {
                startY = e.touches[0].pageY;
            }, { passive: true });
            
            document.addEventListener('touchmove', function(e) {
                if (e.touches[0].pageY > startY && window.pageYOffset === 0) {
                    e.preventDefault();
                }
            }, { passive: false });
        }
        
        // Desktop специфичные оптимизации
        if (!/Mobi|Android/i.test(navigator.userAgent)) {
            console.log('[BalancyWebView] Applying Desktop-specific optimizations');
            
            // Отключение клавиатурных сочетаний
            document.addEventListener('keydown', function(e) {
                // Отключаем Ctrl+A, Ctrl+C, Ctrl+V, Ctrl+S, Ctrl+P и т.д.
                if (e.ctrlKey || e.metaKey) {
                    const blockedKeys = ['a', 'c', 'v', 'x', 's', 'p', 'f', 'h', 'r', 'n', 'w', 't'];
                    if (blockedKeys.includes(e.key.toLowerCase())) {
                        e.preventDefault();
                        e.stopPropagation();
                        return false;
                    }
                }
                
                // Отключаем функциональные клавиши
                const blockedKeyCodes = [
                    112, 113, 114, 115, 116, 117, 118, 119, 120, 121, 122, 123, // F1-F12
                    116, // F5 (Refresh)
                    123  // F12 (Dev Tools)
                ];
                
                if (blockedKeyCodes.includes(e.keyCode)) {
                    e.preventDefault();
                    e.stopPropagation();
                    return false;
                }
            }, { passive: false, capture: true });
            
            // Отключение middle click scroll
            document.addEventListener('mousedown', function(e) {
                if (e.button === 1) { // Middle mouse button
                    e.preventDefault();
                    return false;
                }
            }, { passive: false });
        }
        
        console.log('[BalancyWebView] Game UI behavior applied successfully');
    }
    
    // Функция инициализации
    function initializeGameUI() {
        injectGameUIStyles();
        applyGameUIBehavior();
        
        // Добавляем meta viewport если его нет
        if (!document.querySelector('meta[name="viewport"]')) {
            const viewport = document.createElement('meta');
            viewport.name = 'viewport';
            viewport.content = 'width=device-width, initial-scale=1.0, maximum-scale=1.0, minimum-scale=1.0, user-scalable=no, viewport-fit=cover';
            document.head.appendChild(viewport);
            console.log('[BalancyWebView] Viewport meta tag added');
        }
        
        console.log('[BalancyWebView] Game UI initialization completed');
        
        // Уведомляем о завершении инициализации
        document.dispatchEvent(new CustomEvent('balancyGameUIReady', {
            detail: {
                platform: /iPad|iPhone|iPod/.test(navigator.userAgent) ? 'iOS' : 
                         /Android/.test(navigator.userAgent) ? 'Android' : 'Desktop',
                timestamp: Date.now()
            }
        }));
    }
    
    // Инициализируем когда DOM готов
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initializeGameUI);
    } else {
        initializeGameUI();
    }
    
})();
