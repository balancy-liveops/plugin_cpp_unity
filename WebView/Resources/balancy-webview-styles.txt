    // Function to check if element is scrollable
    function isScrollableElement(element) {
        if (!element || element === document.body || element === document.documentElement) {
            return false;
        }
        
        var style = window.getComputedStyle(element);
        var overflow = style.overflow || style.overflowY || style.overflowX;
        
        return overflow === 'auto' || 
               overflow === 'scroll' || 
               element.classList.contains('scrollable') ||
               element.classList.contains('scroll-container') ||
               element.classList.contains('overflow-auto') ||
               element.classList.contains('overflow-scroll') ||
               element.scrollHeight > element.clientHeight ||
               element.scrollWidth > element.clientWidth;
    }

    // Function to find nearest scrollable parent
    function findScrollableParent(element) {
        var current = element;
        while (current && current !== document.body && current !== document.documentElement) {
            if (isScrollableElement(current)) {
                return current;
            }
            current = current.parentElement;
        }
        return null;
    }

    // Universal JavaScript optimizations
    function applyGameUIBehavior() {
        // Disable context menu
        document.addEventListener('contextmenu', function(e) {
            e.preventDefault();
            e.stopPropagation();
            return false;
        }, { passive: false, capture: true });

        // Disable text selection
        document.addEventListener('selectstart', function(e) {
            e.preventDefault();
            e.stopPropagation();
            return false;
        }, { passive: false, capture: true });

        // Disable drag and drop
        document.addEventListener('dragstart', function(e) {
            e.preventDefault();
            e.stopPropagation();
            return false;
        }, { passive: false, capture: true });

        // iOS specific optimizations
        if (/iPad|iPhone|iPod/.test(navigator.userAgent)) {
            console.log('[BalancyWebView] Applying iOS-specific optimizations');

            // Smart touchmove handling for iOS
            document.addEventListener('touchmove', function(e) {
                // Allow touchmove for input fields
                if (['INPUT', 'TEXTAREA'].includes(e.target.tagName)) {
                    return;
                }
                
                // Find nearest scrollable parent element
                var scrollableParent = findScrollableParent(e.target);
                
                if (scrollableParent) {
                    // If scrollable parent found, allow scrolling
                    return;
                } else {
                    // If no scrollable parent found, prevent page bounce effect
                    e.preventDefault();
                }
            }, { passive: false });

            // Prevent zoom on double tap
            var lastTouchEnd = 0;
            document.addEventListener('touchend', function(event) {
                var now = Date.now();
                if (now - lastTouchEnd <= 300) {
                    event.preventDefault();
                }
                lastTouchEnd = now;
            }, { passive: false });

            // Disable callout
            document.addEventListener('touchstart', function(e) {
                e.target.style.webkitTouchCallout = 'none';
            }, { passive: true });
        }

        // Android specific optimizations
        if (/Android/.test(navigator.userAgent)) {
            console.log('[BalancyWebView] Applying Android-specific optimizations');

            // Prevent system gestures at screen edges
            document.addEventListener('touchstart', function(e) {
                if (e.touches[0].clientX < 20 ||
                    e.touches[0].clientX > window.innerWidth - 20 ||
                    e.touches[0].clientY < 20 ||
                    e.touches[0].clientY > window.innerHeight - 20) {
                    e.preventDefault();
                }
            }, { passive: false });

            // Disable pull-to-refresh only for main page
            var startY = 0;
            document.addEventListener('touchstart', function(e) {
                startY = e.touches[0].pageY;
            }, { passive: true });

            document.addEventListener('touchmove', function(e) {
                // Check if we're inside a scrollable element
                var scrollableParent = findScrollableParent(e.target);
                
                if (!scrollableParent && e.touches[0].pageY > startY && window.pageYOffset === 0) {
                    e.preventDefault();
                }
            }, { passive: false });
        }

        // Desktop specific optimizations
        if (!/Mobi|Android/i.test(navigator.userAgent)) {
            console.log('[BalancyWebView] Applying Desktop-specific optimizations');

            // Disable keyboard shortcuts
            document.addEventListener('keydown', function(e) {
                // Disable Ctrl+A, Ctrl+C, Ctrl+V, Ctrl+S, Ctrl+P, etc.
                if (e.ctrlKey || e.metaKey) {
                    var blockedKeys = ['a', 'c', 'v', 'x', 's', 'p', 'f', 'h', 'r', 'n', 'w', 't'];
                    if (blockedKeys.includes(e.key.toLowerCase())) {
                        e.preventDefault();
                        e.stopPropagation();
                        return false;
                    }
                }

                // Disable function keys
                var blockedKeyCodes = [
                    112, 113, 114, 115, 116, 117, 118, 119, 120, 121, 122, 123, // F1-F12
                    116, // F5 (Refresh)
                    123  // F12 (Dev Tools)
                ];

                if (blockedKeyCodes.includes(e.keyCode)) {
                    e.preventDefault();
                    e.stopPropagation();
                    return false;
                }
            }, { passive: false, capture: true });

            // Disable middle click scroll
            document.addEventListener('mousedown', function(e) {
                if (e.button === 1) { // Middle mouse button
                    e.preventDefault();
                    return false;
                }
            }, { passive: false });
        }

        console.log('[BalancyWebView] Game UI behavior applied successfully');
    }

    // Initialization function
    function initializeGameUI() {
        injectGameUIStyles();
        applyGameUIBehavior();

        // Add meta viewport if it doesn't exist
        if (!document.querySelector('meta[name="viewport"]')) {
            var viewport = document.createElement('meta');
            viewport.name = 'viewport';
            viewport.content = 'width=device-width, initial-scale=1.0, maximum-scale=1.0, minimum-scale=1.0, user-scalable=no, viewport-fit=cover';
            document.head.appendChild(viewport);
            console.log('[BalancyWebView] Viewport meta tag added');
        }

        console.log('[BalancyWebView] Game UI initialization completed');

        // Notify about initialization completion
        document.dispatchEvent(new CustomEvent('balancyGameUIReady', {
            detail: {
                platform: /iPad|iPhone|iPod/.test(navigator.userAgent) ? 'iOS' :
                         /Android/.test(navigator.userAgent) ? 'Android' : 'Desktop',
                timestamp: Date.now()
            }
        }));
    }

    // Initialize when DOM is ready
    if (document.readyState === 'loading') {
        document.addEventListener('DOMContentLoaded', initializeGameUI);
    } else {
        initializeGameUI();
    }
